'没有匹配的日志' : jobLogJob?.status_message || '(暂无日志)'); content.scrollTop = $('#jobLogFollow').checked ? content.scrollHeight : previousScroll; $('#jobLogMatchCount').textContent = `${filtered.length} / ${lines.length} 行 · 复制/下载保留完整日志`; } function openLog(jobId) { activeLogJob = jobId; jobLogGeneration++; jobLogText = ''; jobLogJob = null; for (const id of ['jobLogQuery','jobLogLevel','jobLogPhase']) $('#'+id).value = ''; $('#jobLogMatchCount').textContent = ''; $('#logJobId').textContent = jobId; $('#logPanel').classList.remove('hidden'); updateModalScrollLock(); $('#logContent').textContent = '加载中…'; $('#jobLogStatus').textContent = '加载中…'; $('#jobLogFollow').checked = true; $('#btnCopyJobLog').disabled = true; $('#btnDownloadJobLog').disabled = true; clearInterval(logTimer); logTimer = setInterval(pollLog, 5000); pollLog(); } async function pollLog() { if (activeLogJob == null || jobLogPending === jobLogGeneration) return; const jobId = activeLogJob; const generation = jobLogGeneration; jobLogPending = generation; try { const r = await api(`/api/jobs/${jobId}/log`); if (activeLogJob !== jobId || jobLogGeneration !== generation) return; jobLogText = r.log || ''; jobLogJob = r.job || null; renderJobLog(); $('#btnCopyJobLog').disabled = !jobLogText; $('#btnDownloadJobLog').disabled = !jobLogText; const finished = r.job && ['success','failed','stopped','cancelled','blocked','disabled','partial_success','already_registered'].includes(r.job.status); const explanation = r.job?.status_message || ''; $('#jobLogStatus').textContent = finished ? ('任务已结束' + (explanation ? ' · ' + explanation : '')) : (explanation || '每 5 秒刷新'); if (finished) clearInterval(logTimer); } catch (e) { if (activeLogJob === jobId && jobLogGeneration === generation) { $('#jobLogStatus').textContent = '刷新失败: ' + e.message; } } finally { if (jobLogPending === generation) jobLogPending = null; } } for (const id of ['jobLogQuery','jobLogLevel','jobLogPhase']) { $('#'+id).addEventListener(id === 'jobLogQuery' ?
