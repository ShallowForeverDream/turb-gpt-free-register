'任务已结束' : '每 5 秒刷新'; if (finished) clearInterval(logTimer); } catch (e) { if (activeLogJob === jobId && jobLogGeneration === generation) { $('#jobLogStatus').textContent = '刷新失败: ' + e.message; } } finally { if (jobLogPending === generation) jobLogPending = null; } } $('#btnCopyJobLog').addEventListener('click', () => copyText(jobLogText)); $('#btnDownloadJobLog').addEventListener('click', () => { if (!jobLogText || activeLogJob == null) return; const url = URL.createObjectURL(new Blob([jobLogText], {type: 'text/plain;charset=utf-8'})); const link = document.createElement('a'); link.href = url; link.download = `job-${activeLogJob}.log`; document.body.appendChild(link); link.click(); link.remove(); setTimeout(() => URL.revokeObjectURL(url), 1000); }); $('#jobLogFollow').addEventListener('change', () => { if ($('#jobLogFollow').checked) $('#logContent').scrollTop = $('#logContent').scrollHeight; }); $('#logContent').addEventListener('scroll', () => { const c = $('#logContent'); if (c.scrollTop + c.clientHeight < c.scrollHeight - 30) $('#jobLogFollow').checked = false; });
