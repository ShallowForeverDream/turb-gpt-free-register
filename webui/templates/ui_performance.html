A slow request must // not grow an interval backlog, and actions must not create extra poll loops. function pollVisiblePage(tab, callback, interval) { let running = false; const tick = async () => { if (document.hidden || running) return; if (tab && document.getElementById('tab-'+tab)?.classList.contains('hidden')) return; running = true; try { await callback(); } finally { running = false; } }; const timer = setInterval(tick, interval); document.addEventListener('visibilitychange', () => { if (!document.hidden) tick(); }); return timer; }
